cmake_minimum_required(VERSION 3.22.1)
project(anlr_native LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -O2)
add_library(anlr_pty SHARED pty_jni.cpp)
target_compile_features(anlr_pty PRIVATE cxx_std_17)
target_compile_options(anlr_pty PRIVATE -Wall -Wextra -Werror)
find_library(log-lib log)
find_library(android-lib android)
find_library(aaudio-lib aaudio)
target_link_libraries(anlr_pty ${log-lib})
add_library(anlr_display SHARED display_bridge.c)
target_link_libraries(anlr_display ${android-lib} ${log-lib})
add_library(anlr_io SHARED x11_input.c audio_bridge.c)
target_link_libraries(anlr_io ${aaudio-lib} ${log-lib})
